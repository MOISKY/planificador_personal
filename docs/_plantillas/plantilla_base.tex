\newcommand{\infonote}[2]{%

  left=8pt,right=8pt,top=4pt,bottom=4pt,fonttitle=\bfseries\color{white},
  colbacktitle=noilblue,title={#1}]

\end{tcolorbox}\par\addvspace{4pt}}

\par\addvspace{4pt}\noindent

  colbacktitle=mPut,title={#1}]
\small #2
\end{tcolorbox}\par\addvspace{4pt}}
